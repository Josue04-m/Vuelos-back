internal sealed class PayPalStorePrueba(EstadoReservasPrueba f) : IPagoPayPalRepository
{
    public readonly HashSet<string> Eventos = [];
    public Task<PagoPayPal?> PorReservaAsync(int id, CancellationToken ct) => Task.FromResult(f.Pagos.SingleOrDefault(p => p.ReservaId == id));
    public Task<PagoPayPal?> PorIdAsync(Guid id, CancellationToken ct) => Task.FromResult(f.Pagos.SingleOrDefault(p => p.PagoId == id));
    public Task<PagoPayPal?> PorIdentificadorExternoAsync(string orden, string captura, string entorno, CancellationToken ct) =>
        Task.FromResult(f.Pagos.SingleOrDefault(p => p.Entorno == entorno && (orden != "" && p.OrdenId == orden || captura != "" && p.CapturaId == captura)));
    public void Agregar(PagoPayPal p) { p.Reserva = f.Reservas.Single(r => r.ReservaId == p.ReservaId); f.Pagos.Add(p); }
    public void LimpiarSeguimiento() { }
    public Task<bool> EventoExisteAsync(string id, CancellationToken ct) => Task.FromResult(Eventos.Contains(id));
    public Task RegistrarEventoAsync(EventoPayPal e, CancellationToken ct) { Eventos.Add(e.EventoId); return Task.CompletedTask; }
}

internal sealed class ProveedorPayPalPrueba(EstadoReservasPrueba f) : IPayPalClient
{
    public int Creaciones, Capturas;
    public bool Aprobada = true, Completada, Pendiente, FallarCaptura, FirmaValida = true, Disponible = true;
    public Action? DuranteCaptura;
    private PagoPayPal? pago;
    private OrdenPayPal Orden() => new("ORDER123", Completada ? "COMPLETED" : Aprobada ? "APPROVED" : "CREATED",
        "https://www.sandbox.paypal.com/checkoutnow?token=ORDER123", pago!.PagoId.ToString("N"), pago.Monto, "USD", pago.MerchantId,
        Completada || Pendiente && Capturas > 0 ? "CAPTURE123" : null, Completada ? "COMPLETED" : Pendiente && Capturas > 0 ? "PENDING" : null,
        Completada || Pendiente && Capturas > 0 ? pago.Monto : null, Completada || Pendiente && Capturas > 0 ? "USD" : null,
        Completada ? f.Clock.GetUtcNow().UtcDateTime : null);
    private void SinTransaccion() { if (f.EnTransaccion) throw new Exception("HTTP dentro de transacción SQL"); if (!Disponible) throw new ReservaOperacionException(ErrorReserva.ServicioNoDisponible,"proveedor_caido","Proveedor no disponible"); }
    public Task<OrdenPayPal> CrearAsync(PagoPayPal p, CancellationToken ct) { SinTransaccion(); pago = p; Creaciones++; return Task.FromResult(Orden()); }
    public Task<OrdenPayPal> ConsultarAsync(string id, CancellationToken ct) { SinTransaccion(); return Task.FromResult(Orden()); }
    public Task<OrdenPayPal> CapturarAsync(PagoPayPal p, CancellationToken ct)
    {
        SinTransaccion(); if (!p.BloqueaReserva || !p.CapturaIniciada) throw new Exception("Captura sin protección durable");
        Capturas++; DuranteCaptura?.Invoke(); if (!Pendiente) Completada = true;
        if (FallarCaptura) throw new ReservaOperacionException(ErrorReserva.ServicioNoDisponible,"captura_incierta","Respuesta incierta");
        return Task.FromResult(Orden());
    }
    public Task<bool> VerificarEventoAsync(JsonElement e, IHeaderDictionary h, CancellationToken ct) => Task.FromResult(FirmaValida);
}

internal sealed class HttpPayPalPrueba : HttpMessageHandler
{
    public PagoPayPal Pago=null!;
    public readonly List<string> CreateIds=[],CaptureIds=[],Hosts=[];
    public string? CreateBody;
    public bool Fallar;
    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage req,CancellationToken ct)
    {
        Hosts.Add(req.RequestUri!.Host);
        if(Fallar) return new(HttpStatusCode.ServiceUnavailable){Content=new StringContent("provider private error must not leak")};
        var path=req.RequestUri.AbsolutePath;
        if(path.EndsWith("/token")) return Json("""{"access_token":"fake-access-token"}""");
        if(path=="/v2/checkout/orders") {CreateIds.Add(req.Headers.GetValues("PayPal-Request-Id").Single());CreateBody=await req.Content!.ReadAsStringAsync(ct);}
        if(path.EndsWith("/capture")) CaptureIds.Add(req.Headers.GetValues("PayPal-Request-Id").Single());
        return Json(JsonSerializer.Serialize(new {id="ORDER123",status="APPROVED",purchase_units=new[]{new {custom_id=Pago.PagoId.ToString("N"),
            amount=new {value=Pago.Monto.ToString("0.00",System.Globalization.CultureInfo.InvariantCulture),currency_code="USD"},payee=new {merchant_id=Pago.MerchantId}}}}));
    }
    private static HttpResponseMessage Json(string json)=>new(HttpStatusCode.OK){Content=new StringContent(json,Encoding.UTF8,"application/json")};
}
