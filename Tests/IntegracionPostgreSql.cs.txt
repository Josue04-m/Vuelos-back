    public static async Task<int> Main()
    {
        var supplied=Environment.GetEnvironmentVariable("PNA_TEST_CONNECTION");
        if(string.IsNullOrWhiteSpace(supplied)||Environment.GetEnvironmentVariable("PNA_TESTS_CONFIRM")!="BASE_DE_PRUEBAS")
        {
            Report("NO_EJECUTADA","falta_conexion_o_confirmacion_de_base_pruebas");
            Console.WriteLine("NO EJECUTADA: configurar PNA_TEST_CONNECTION y PNA_TESTS_CONFIRM=BASE_DE_PRUEBAS. No se abrió ninguna conexión.");
            return 2;
        }
        using var timeout=new CancellationTokenSource(TimeSpan.FromMinutes(4));ct=timeout.Token;
        try
        {
            var builder=new NpgsqlConnectionStringBuilder(supplied){Timeout=10,CommandTimeout=20};
            Check(builder.SslMode==SslMode.VerifyFull,"tls_debe_validar_servidor");connection=builder.ConnectionString;
            await Preflight();
            var fixture=await Prepare();
            await Authentication(fixture);
            await VersionsAndRaces(fixture);
            await ReservationAndPayments(fixture);
            Report("APROBADA");
            Console.WriteLine($"OK: {checks} comprobaciones contra PostgreSQL. Ejecución LAB-{run}; datos de laboratorio conservados. Pago simulado, no PayPal Sandbox real.");
            return 0;
        }
        catch(Exception error)
        {
            // No imprimir mensajes del driver, detalles SQL, passwords, tokens ni conexiones.
            var code=error is Failure ? error.Message : error is PostgresException pg ? "postgres_"+pg.SqlState : error.GetType().Name;
            Report("FALLIDA",code);Console.Error.WriteLine($"FALLO {code}; ejecución LAB-{run}. Revisar sus datos de laboratorio antes de repetir.");return 1;
        }
    }

    private static async Task<T> Scalar<T>(AppDbContext db,string sql)
    {
        if(db.Database.GetDbConnection().State!=ConnectionState.Open)await db.Database.OpenConnectionAsync(ct);
        databaseOpened=true;
        await using var command=db.Database.GetDbConnection().CreateCommand();command.CommandText=sql;
        var value=await command.ExecuteScalarAsync(ct);return (T)Convert.ChangeType(value!,typeof(T));
    }
    private static async Task Preflight()
    {
        await using var db=Db();
        Check(await Scalar<long>(db,"SELECT count(*) FROM pg_catalog.pg_tables WHERE schemaname='pna' AND tablename <> '__EFMigrationsHistory'")==15,"esquema_no_preparado");
        Check(await Scalar<long>(db,"SELECT count(*) FROM pg_catalog.pg_trigger t JOIN pg_catalog.pg_class c ON c.oid=t.tgrelid JOIN pg_catalog.pg_namespace n ON n.oid=c.relnamespace JOIN pg_catalog.pg_proc p ON p.oid=t.tgfoid WHERE n.nspname='pna' AND NOT t.tgisinternal AND p.proname='generar_version_fila' AND t.tgenabled='O'")==7,"triggers_no_preparados");
        Check(!await Scalar<bool>(db,"SELECT has_schema_privilege(current_user,'pna','CREATE')"),"conexion_no_es_runtime_restringido");
        Check(!await Scalar<bool>(db,"SELECT has_table_privilege(current_user,'pna.reservas','DELETE') OR has_table_privilege(current_user,'pna.reservas','TRUNCATE') OR has_table_privilege(current_user,'pna.\"__EFMigrationsHistory\"','UPDATE')"),"runtime_con_permisos_excesivos");
        Check(await Scalar<bool>(db,"SELECT has_table_privilege(current_user,'pna.reservas','INSERT') AND has_table_privilege(current_user,'pna.reservas','UPDATE') AND has_sequence_privilege(current_user,'pna.row_version_seq','USAGE') AND has_function_privilege(current_user,'pna.generar_version_fila()','EXECUTE')"),"runtime_sin_permisos_necesarios");
        await using var second=Db();await second.Database.OpenConnectionAsync(ct);
        Check(((NpgsqlConnection)db.Database.GetDbConnection()).ProcessID!=((NpgsqlConnection)second.Database.GetDbConnection()).ProcessID,"faltan_conexiones_independientes");
    }

    private static async Task ReservationAndPayments(Fixture f)
    {
        var request=await Request(f,f.Seats[0]);var gate=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        async Task<(int User,ReservaResponse? Hold)> Reserve(int user)
        {
            await gate.Task;await using var db=Db();try{return(user,await Reservations(db).CrearAsync(user,request,ct));}
            catch(ReservaOperacionException e)when(e.Error==ErrorReserva.Conflicto){return(user,null);}
        }
        var a=Reserve(f.A);var b=Reserve(f.B);gate.SetResult();var results=await Task.WhenAll(a,b);Check(results.Count(r=>r.Hold!=null)==1,"doble_reserva_concurrente");var winner=results.Single(r=>r.Hold!=null);
        await using(var db=Db())
        {
            Check(await db.DetallesReserva.CountAsync(d=>d.AsientoVueloId==f.Seats[0] && d.Estado=="activo",ct)==1,"asignacion_activa_unica");
            Check(winner.Hold!.Subtotal==100 && winner.Hold.MontoIva==15 && winner.Hold.MontoTotal==115,"iva_persistido");
            await using(var tx=await db.Database.BeginTransactionAsync(ct))
            {
                var extra=new DetalleReserva{ReservaId=winner.Hold.ReservaId,AsientoVueloId=f.Seats[0],PrimerNombre="LAB",PrimerApellido=run,DocumentoIdentidad="0987654321",TipoPasajero="adulto",PrecioCongelado=100,Estado="activo"};
                db.DetallesReserva.Add(extra);var unique=false;
                try{await db.SaveChangesAsync(ct);}catch(DbUpdateException e)when(e.InnerException is PostgresException pg && pg.SqlState=="23505" && pg.ConstraintName=="UX_detalle_reservas_asiento_activo"){unique=true;}
                await tx.RollbackAsync(ct);db.Entry(extra).State=EntityState.Detached;
                Check(unique,"restriccion_unica_activa_real");
            }
            var other=winner.User==f.A?f.B:f.A;var rejected=false;try{await Reservations(db).ObtenerPropiaAsync(other,winner.Hold.ReservaId,ct);}catch(ReservaOperacionException e)when(e.Error==ErrorReserva.NoEncontrada){rejected=true;}Check(rejected,"reserva_ajena_oculta");
            await Reservations(db).CancelarAsync(winner.User,winner.Hold.ReservaId,new(){RowVersion=winner.Hold.RowVersion},ct);
        }
        await using(var db=Db())
        {
            Check((await db.AsientosVuelo.SingleAsync(s=>s.AsientoVueloId==f.Seats[0],ct)).Estado=="disponible","cancelacion_libera");
            var hold=await Reservations(db).CrearAsync(f.A,await Request(f,f.Seats[0]),ct);clock.Advance(TimeSpan.FromHours(13));
            var repo=new ReservaRepository(db);await new ExpiracionReservasService(repo,new CicloReservaRepository(db,clock),clock).ExpirarVueloAsync(f.Flight,ct);
            Check((await db.Reservas.AsNoTracking().SingleAsync(r=>r.ReservaId==hold.ReservaId,ct)).Estado=="expirada","expiracion_persistida");
        }
        var external=new ExternalState();ReservaResponse payable;
        await using(var db=Db()){payable=await Reservations(db).CrearAsync(f.A,await Request(f,f.Seats[0]),ct);await Payment(db,external).CrearAsync(f.A,payable.ReservaId,new(){RowVersion=payable.RowVersion},ct);}
        var start=new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);
        async Task Capture(){await start.Task;await using var db=Db();await Payment(db,external).CapturarAsync(f.A,payable.ReservaId,ct);}
        var one=Capture();var two=Capture();start.SetResult();await Task.WhenAll(one,two);
        await using(var db=Db())
        {
            var paid=await Payment(db,external).ConsultarAsync(f.A,payable.ReservaId,ct);
            Check(paid.EstadoReserva=="pagada" && external.Captures==1 && paid.Monto==115,"captura_real_db_idempotente_proveedor_simulado");
            var eventId="LAB-"+run;var store=new PagoPayPalRepository(db);
            await store.RegistrarEventoAsync(new(){EventoId=eventId,Tipo="LAB_TEST",FechaProcesado=clock.GetUtcNow().UtcDateTime},ct);
            await store.RegistrarEventoAsync(new(){EventoId=eventId,Tipo="LAB_TEST",FechaProcesado=clock.GetUtcNow().UtcDateTime},ct);
            Check(await db.EventosPayPal.CountAsync(e=>e.EventoId==eventId,ct)==1,"evento_idempotente_en_db_no_prueba_firma");
        }
        var pending=new ExternalState{Pending=true};ReservaResponse waiting;
        await using(var db=Db())
        {
            waiting=await Reservations(db).CrearAsync(f.B,await Request(f,f.Seats[2]),ct);
            await Payment(db,pending).CrearAsync(f.B,waiting.ReservaId,new(){RowVersion=waiting.RowVersion},ct);
            await Payment(db,pending).CapturarAsync(f.B,waiting.ReservaId,ct);
            clock.Advance(TimeSpan.FromHours(13));var repo=new ReservaRepository(db);
            await new ExpiracionReservasService(repo,new CicloReservaRepository(db,clock),clock).ExpirarVueloAsync(f.Flight,ct);
            Check((await db.Reservas.AsNoTracking().SingleAsync(r=>r.ReservaId==waiting.ReservaId,ct)).Estado=="pendiente_pago","pago_pendiente_no_expira");
            var state=await Payment(db,pending).ConsultarAsync(f.B,waiting.ReservaId,ct);pending.Completed=true;
            await Payment(db,pending).ConciliarAsync(state.PagoId,ct);
            Check((await db.Reservas.AsNoTracking().SingleAsync(r=>r.ReservaId==waiting.ReservaId,ct)).Estado=="pagada" && pending.Captures==1,"conciliacion_sin_recobro");
        }
    }

    private static PagoPayPalService Payment(AppDbContext db,ExternalState state)=>new(new PagoPayPalRepository(db),new ReservaRepository(db),new ExternalClient(db,state),Options.Create(new OpcionesPayPal{Habilitado=true,Entorno="Sandbox",ClientId="fake-client",ClientSecret="fake-secret",MerchantId="SELLER123",WebhookId="LABWEBHOOK",ReturnUrl="https://frontend.example/retorno",CancelUrl="https://frontend.example/cancelar"}),clock);
    private sealed class ExternalState{public PagoPayPal? Order;public int Captures;public bool Pending,Completed;}
    private sealed class ExternalClient(AppDbContext db,ExternalState state):IPayPalClient
    {
        private void Guard(){if(db.Database.CurrentTransaction!=null)throw new Failure("proveedor_dentro_transaccion");}
        private OrdenPayPal Result()
        {
            var p=state.Order!;var captured=state.Captures>0;
            return new("LABORDER"+p.PagoId.ToString("N"),state.Completed?"COMPLETED":"APPROVED","https://sandbox.example.invalid/aprobar",p.PagoId.ToString("N"),p.Monto,"USD",p.MerchantId,captured?"LABCAP"+p.PagoId.ToString("N"):null,captured?(state.Completed?"COMPLETED":"PENDING"):null,captured?p.Monto:null,captured?"USD":null,state.Completed?clock.GetUtcNow().UtcDateTime:null);
        }
        public Task<OrdenPayPal> CrearAsync(PagoPayPal payment,CancellationToken token){Guard();state.Order=payment;return Task.FromResult(Result());}
        public Task<OrdenPayPal> ConsultarAsync(string id,CancellationToken token){Guard();return Task.FromResult(Result());}
        public Task<OrdenPayPal> CapturarAsync(PagoPayPal payment,CancellationToken token){Guard();Interlocked.Increment(ref state.Captures);if(!state.Pending)state.Completed=true;return Task.FromResult(Result());}
        public Task<bool> VerificarEventoAsync(JsonElement value,IHeaderDictionary headers,CancellationToken token)=>Task.FromResult(false);
    }
    private sealed class ReservationOptions:IOptionsMonitor<OpcionesReservas>
    {
        public OpcionesReservas CurrentValue{get;}=new(){PorcentajeIva=15};public OpcionesReservas Get(string? name)=>CurrentValue;
        public IDisposable? OnChange(Action<OpcionesReservas,string?> listener)=>null;
    }
    private sealed class Clock:TimeProvider
    {
        private long ticks=DateTimeOffset.UtcNow.UtcTicks/10*10;
        public override DateTimeOffset GetUtcNow()=>new(new DateTime(Interlocked.Read(ref ticks),DateTimeKind.Utc));
        public void Advance(TimeSpan value)=>Interlocked.Add(ref ticks,value.Ticks);
    }
